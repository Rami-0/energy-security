/**
 * Функция расчета вероятности остановки техпроцесса
 * @param {number} uOst - Остаточное напряжение в % (от 0 до 100)
 * @param {number} duration - Длительность провала в мс
 * @param {string} loadType - Тип нагрузки (из select)
 * @returns {object} Объект с результатами и промежуточными коэффициентами
 */
function calculateSagRisk(uOst, duration, loadType) {
    // 1. База знаний оборудования (Эмпирические константы)
    const loadProfiles = {
        "Насосная станция (АД 6кВ)": { 
            uCrit: 30,  // Критический порог ЗМН/контактора (%)
            tCrit: 200, // Критическое время (мс)
            pb: 40      // Базовая вероятность сбоя (%)
        },
        "Компрессорная установка (ЧРП)": { 
            uCrit: 15,  // ЧРП очень чувствительны к падению напряжения (допуск 10-15%)
            tCrit: 50,  // Быстро отключаются по защите звена постоянного тока
            pb: 80 
        },
        "Система управления (ПЛК)": { 
            uCrit: 40,  // Блоки питания ПЛК имеют запас емкости
            tCrit: 20,  // Но если емкость исчерпана, сбой мгновенный (1-2 периода сети)
            pb: 90 
        },
        "Цех ЭЛОУ-АВТ (Комплекс)": { 
            uCrit: 30,  
            tCrit: 150, 
            pb: 60      // Комплексная оценка смешанной нагрузки
        }
    };

    // Получаем профиль оборудования (или берем дефолтный)
    const profile = loadProfiles[loadType] || loadProfiles["Насосная станция (АД 6кВ)"];

    // 2. Расчет глубины провала (dU)
    const dU = 100 - parseFloat(uOst);

    // Если провала нет (напряжение 100%), риск равен 0
    if (dU <= 0) {
        return { probability: 0, dU: 0, Ku: 0, Kt: 0, Pb: profile.pb };
    }

    // 3. Расчет коэффициентов чувствительности
    const Ku = dU / profile.uCrit;
    const Kt = parseFloat(duration) / profile.tCrit;

    // 4. Расчет итоговой вероятности
    let P = profile.pb * Ku * Kt;

    // 5. Нормализация (вероятность не может быть < 0% или > 100%)
    P = Math.max(0, Math.min(100, P));

    // Возвращаем объект со всеми данными для вывода на экран
    return {
        probability: P.toFixed(1), // Итоговый процент с 1 знаком после запятой
        dU: dU.toFixed(1),
        Ku: Ku.toFixed(2),
        Kt: Kt.toFixed(2),
        Pb: profile.pb
    };
}
